## 9. 复现命令

```bash
# 1) 重建 registry（可删库重来）
python3 tools/p20/p20ctl.py init --force
python3 tools/p20/p20ctl.py validate

# 2) 查看配置与兼容性
python3 tools/p20/p20ctl.py list workflows
python3 tools/p20/p20ctl.py show workflow qwen_image_t2i
python3 tools/p20/p20ctl.py compat list

# 3) 解析/预演（不花钱）
python3 tools/p20/p20ctl.py resolve --workflow qwen_image_t2i \
  --set prompt="a red apple" --set seed=20261102

# 4) 建作业并真跑（会花钱）
python3 tools/p20/p20ctl.py new --workflow qwen_image_t2i --model qwen_image \
  --set prompt="a red apple" --set seed=20261102 --run --gpu 4090

# 5) 零代码验证（只加配置后）
diff <(sort reports/p2/p20/task5b-core-before.sha256) \
     <(sort reports/p2/p20/task5b-core-after.sha256)   # 空 = 零差异

# 6) 回归
python3 -m unittest discover -s tools/p20/tests -t .
python3 -m unittest discover -s tools/p2/tests -t .
python3 tools/p1/cases.py
python3 tools/p1/jobctl.py validate --deep
```
